use std::{io::{self,Write},thread,time::Duration};
fn main() {
    for step in 1..=2 {
        write!(io::stdout(),"STDOUT_{step}").unwrap();
        io::stdout().flush().unwrap();
        write!(io::stderr(),"STDERR_{step}").unwrap();
        io::stderr().flush().unwrap();
        thread::sleep(Duration::from_millis(1000));
    }
}
